\subsection{Séries à termes positifs}

\begin{definition}[Série à termes positifs (atp)]
    On appelle série à termes positifs (atp) toute série numérique 
    dont les termes sont positifs à partir d'un certain rang, c'est-à-dire

    Soit $\left(\sum\limits_{n=0}^{+\infty} u_n \right)$, on dit que c'est une série atp si
    $$
    \exists N \in \mathbb{N}, \forall n \geq N, u_n \geq 0
    $$
\end{definition}

\begin{remarque}
    Comme on l'a vu l'ensemble des séries convergentes est un espace vectoriel,
    donc en multipliant par -1 une série à termes négatifs on obtient une série à termes positifs.
\end{remarque}

\begin{theoreme}{Critère de comparaison pour les séries atp}{}
    Soit $\left( \sum\limits_{n=0}^{+\infty} u_n \right)$ et $\left( \sum\limits_{n=0}^{+\infty} v_n \right)$ deux séries atp telles que
    $$
    \exists N \in \mathbb{N}, \forall n \geq N, u_n \leq v_n
    $$
    Alors
    \begin{enumerate}
        \item Si $\left( \sum\limits_{n=0}^{+\infty} v_n \right)$ converge alors $\left( \sum\limits_{n=0}^{+\infty} u_n \right)$ converge.
        \item Si $\left( \sum\limits_{n=0}^{+\infty} u_n \right)$ diverge alors $\left( \sum\limits_{n=0}^{+\infty} v_n \right)$ diverge.
    \end{enumerate}
\end{theoreme}

\begin{demonstration}
    On a par hypothèse $\forall n \in \N, u_n \leq v_n$ (en négligeant les premiers termes).
    \begin{enumerate}
        \item Si $\left( \sum\limits_{n=0}^{+\infty} v_n \right)$ converge alors $\left( S_n(v) \right)$ converge et est majorée par sa limite (car c'est une suite croissante).
        Comme $\left( S_n(u) \right)$ est croissante et majorée par $\left( S_n(v) \right)$, elle converge également.
        \item Si $\left( \sum\limits_{n=0}^{+\infty} u_n \right)$ diverge alors $\left( S_n(u) \right)$ diverge vers $+\infty$ (car c'est une suite croissante).
        Comme $\left( S_n(v) \right)$ est croissante et minorée par $\left( S_n(u) \right)$, elle diverge également vers $+\infty$.
    \end{enumerate}
\end{demonstration}

\begin{exemple}
    Étudier la convergence de la série $\left(\sum\limits_{n=1}^{+\infty} \dfrac{1}{n^2 + n + 1}\right)$.
    \newline
    On a $0 \leq \dfrac{1}{n^2 + n + 1} \leq \dfrac{1}{n^2}$
    Or la série $\sum\limits_{n=1}^{+\infty} \dfrac{1}{n^2}$ est une série de Riemann convergente (car $2 \gt 1$).
    Donc par le critère de comparaison, la série $\sum\limits_{n=1}^{+\infty} \dfrac{1}{n^2 + n + 1}$ converge également.
\end{exemple}

\begin{remarque}
    Le critère ne nous dit rien sur la somme de la série, ça nous dit juste si elle converge ou diverge.
\end{remarque}

\begin{methode}
    Quand on a une une série à terme général $u_n$ qui tend vers 0, et que le produit par $n^\alpha$ tend vers 0 aussi (par exemple dans le cas d'une croissance comparée avec un exponentielle), alors la série converge.

    En effet, $n^\alpha u_n = \dfrac{u_n}{\frac{1}{n^\alpha}} \underset{n \rightarrow +\infty}{\longrightarrow} 0$ et $\dfrac{1}{n^\alpha}$ est le terme général d'une série de Riemann convergente, donc par le critère de comparaison, la série de terme général $u_n$ converge.
\end{methode}

\begin{definition}
    Deux suites $\left( u_n \right)$ et $\left( v_n \right)$ sont dites équivalentes si
    $$
    \exists \left( w_n \right) \in \R^{\N}, w_n \rightarrow 1, u_n = v_n w_n
    $$
\end{definition}

\begin{theoreme}{Critère par équivalents pour les séries atp}{}
    Soit $\left( \sum\limits_{n=0}^{+\infty} u_n \right)$ et $\left( \sum\limits_{n=0}^{+\infty} v_n \right)$ deux séries atp telles que
    $$
    u_n \sim v_n
    $$
    Alors les deux séries sont de même nature (convergentes ou divergentes en même temps).

    De plus, si les deux séries convergent alors
    $$
    \sum\limits_{k=n+1}^{+\infty} u_k \sim \sum\limits_{k=n+1}^{+\infty} v_k
    $$
    et si les deux séries divergent alors
    $$
    \sum\limits_{k=0}^{n} u_k \sim \sum\limits_{k=0}^{n} v_k
    $$
\end{theoreme}

\begin{demonstration}
    On suppose que $u_n \sim v_n$ et que $v_n \geq 0$ pour $n$ assez grand.

    Comme $u_n \sim v_n$, on a aussi $\lim \dfrac{u_n}{v_n} = 1$,
    donc il existe un $n_0$ tel que pour $n \geq n_0, \quad \frac{1}{2} v_n \leq u_n \leq 2 v_n$.

    \begin{description}
        \item[Si $v_n$ converge.]
        Par le critère de comparaison, la série de terme général $u_n$ converge également.

        De plus, $u_n \sim v_n \iff \forall \varepsilon > 0, \exists N \in \N, \forall n \geq N, (1 - \varepsilon) v_n \leq u_n \leq (1 + \varepsilon) v_n$.

        En sommant pour $n$ allant de $N$ à $+\infty$, on a
        $$
        (1 - \varepsilon) \sum\limits_{k=n+1}^{+\infty} v_k \leq \sum\limits_{k=n+1}^{+\infty} u_k \leq (1 + \varepsilon) \sum\limits_{k=n+1}^{+\infty} v_k
        $$

        Donc $\sum\limits_{k=n+1}^{+\infty} u_k \sim \sum\limits_{k=n+1}^{+\infty} v_k$.
        
        \item[Si $v_n$ diverge.]
        Par le critère de comparaison, la série de terme général $u_n$ diverge également.
        \begin{hotwarn}
            Trop de calculs à faire, je ne les fais pas.
        \end{hotwarn}
    \end{description}
\end{demonstration}

\begin{exemple}
    Étudier la convergence de la série $\left( \sum\limits_{n=1}^{+\infty} \ln\left( \dfrac{n+1}{n} \right) \right)$.
    \newline
    On a $\ln\left( \dfrac{n+1}{n} \right) = \ln\left( 1 + \dfrac{1}{n} \right) \sim \dfrac{1}{n}$.
    Or la série $\left(\sum\limits_{n=1}^{+\infty} \dfrac{1}{n}\right)$ est la série harmonique qui est divergente.
    Donc par le critère par équivalents, la série $\sum\limits_{n=1}^{+\infty} \ln\left( \dfrac{n+1}{n} \right)$ diverge également.
\end{exemple}

\begin{theoreme}{Critère de négligeabilité}{}
    Soit $\left( \sum\limits_{n=0}^{+\infty} u_n \right)$ et $\left( \sum\limits_{n=0}^{+\infty} v_n \right)$ deux séries atp telles que
    $$
    u_n = o(v_n)
    $$
    Alors,

    \begin{itemize}
        \item Si $\sum\limits_n v_n$ converge alors $\sum\limits_{n} u_n$ converge et $\sum\limits_{k = n + 1}^{+\infty} u_k = o\left( \sum\limits_{k = n + 1}^{+\infty} v_k \right)$.
        \item Si $\sum\limits_n v_n$ diverge alors $\sum\limits_{k=0}^{n} u_k = o\left( \sum\limits_{k=0}^{n} v_k \right)$.
    \end{itemize}
\end{theoreme}

\begin{demonstration}
    On suppose que $u_n = o(v_n)$ et que $v_n \geq 0$ pour $n$ assez grand.

    Comme $u_n = o(v_n)$, on a aussi $\lim \dfrac{u_n}{v_n} = 0$,
    donc il existe un $n_0$ tel que pour $n \geq n_0, \quad u_n \leq v_n$.

    \begin{description}
        \item[Si $v_n$ converge.]
        Par le critère de comparaison, la série de terme général $u_n$ converge également.

        De plus, $u_n = o(v_n) \iff \forall \varepsilon > 0, \exists N \in \N, \forall n \geq N, u_n \leq \varepsilon v_n$.

        En sommant pour $n$ allant de $N$ à $+\infty$, on a
        $$
        0 \leq \sum\limits_{k=n+1}^{+\infty} u_k \leq \varepsilon \sum\limits_{k=n+1}^{+\infty} v_k
        $$

        Donc $\sum\limits_{k=n+1}^{+\infty} u_k = o\left( \sum\limits_{k=n+1}^{+\infty} v_k \right)$.
        
        \item[Si $v_n$ diverge.]\n
        \begin{hotwarn}
            Trop de calculs à faire, je ne les fais pas.
        \end{hotwarn}
    \end{description}
\end{demonstration}

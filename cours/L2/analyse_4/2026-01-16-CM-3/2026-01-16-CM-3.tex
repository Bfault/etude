\begin{proposition}{Dérivabilité}{}
    Soit $I$ un segment de $\R$ et $(f_n)$ une suite de fonctions de $I$ dans $\R$.

    On suppose que $\forall n \in \N, f_n$ est $C^1$ sur $I$ et il existe $g: I \to \R$ telle que
    \begin{itemize}[label=--]
        \item $(f_n)$ converge simplement vers $f$ sur un point $x_0 \in I$;
        \item $(f_n')$ converge uniformément vers $g$ sur $I$.
    \end{itemize}
    Alors $(f_n)$ converge uniformément vers $f$ sur $I$ et $f$ est $C^1$ et $f' = g$.
\end{proposition}

\begin{demonstration}
    Pour tout $n \in \N$ on pose $\funcdef{G_n}{I}{\R}{x}{\int_{x_0}^x f_n'(t)dt}$ et $\funcdef{G}{I}{\R}{x}{\int_{x_0}^x g(t)dt}$.

    En appliquant le théorème \ref{theoreme:integrale} à $(f_n')$, on obtient que $G_n$ converge uniformément vers $G$ sur $I$.

    Par ailleurs, d'après le théorème fondamental de l'analyse, on a d'une part $G_n(x) = f_n(x) - f_n(x_0)$ donc $(f_n - f_n(x_0))$ converge uniformément vers $G$ sur $I$.
    Par hypothèse, $(f_n(x_0))$ converge simplement et $\lim\limits_{n \to +\infty} f_n(x_0) = l$.
    D'autre part, $f_n(x) = G_n(x) + f_n(x_0)$ donc en passant à la limite, on obtient que $f(x) = G(x) + l$.

    Ainsi, $(f_n)$ converge uniformément vers $f$ sur $I$ par somme de fonctions uniformément convergentes sur $I$.

    $g$ est continue par le théorème \ref{theoreme:continuite} (limite uniforme de fonctions continues $f_n'$). Ainsi $G$ est de classe $C^1$ sur $I$ et $l$ aussi (constante) donc $f$ est $C^1$ sur $I$ et $f' = g$.

\end{demonstration}

\begin{proposition}{Formule de changement de variable}{}
    On prend la série entière $f(X) := \sum\limits_{k \geq k_0} a_k X^k$ de rayon de convergence $R > 0$ et $X \in D_R$.

    Alors pour tout $\alpha \in \mathbb{C}$, pour tout $m \in \N^*$, on a
    $$
    \sum_{k \geq k_0} a_k X^k = \sum_{k \geq k_0} a_k (\alpha z^m)^k = f(\alpha z^m)
    $$
    dés que $|\alpha z^m| < R \iff |z| < \sqrt[m]{\dfrac{R}{|\alpha|}}$
\end{proposition}

\begin{exemple}
    On sait que $\dfrac{1}{1 - z} = \sum_{n \geq 0} z^n$ pour $|z| < 1$.

    En appliquant la formule de changement de variable, on trouve que
    $$
    \dfrac{1}{1 - z^2} = \sum_{n \geq 0} z^{2n}
    $$
    pour $|z| < \sqrt[2]{\dfrac{1}{|1|}} = 1$.
\end{exemple}

\begin{proposition}{}{}
    Pour tout $a, b \in \mathbb{C}$
    $$
    e^{a + b} = e^a e^b
    $$
\end{proposition}

\begin{demonstration}
    On a
    $$
    e^a = \sum_{n \geq 0} \dfrac{a^n}{n!} = \sum_{n \geq 0} 1^n \dfrac{a^n}{n!}
    \quad \text{et} \quad
    e^b = \sum_{n \geq 0} 1^n \dfrac{b^n}{n!}
    $$

    On fait le produit des deux séries
    $$
    e^a e^b = \left(\sum_{n \geq 0} \dfrac{a^n}{n!} 1^n \right)\left( \sum_{n \geq 0} \dfrac{b^n}{n!} 1^n \right) = \sum_{n \geq 0} c_n 1^n
    $$
    avec
    \begin{align*}
        c_n := \sum_{k = 0}^n \dfrac{a^k}{k!} \dfrac{b^{n-k}}{(n-k)!} &= \dfrac{b^n}{n!} + \dfrac{ab^{n-1}}{1!(n-1)!} + \cdots + \dfrac{a^n}{n!} \\
        &= \dfrac{1}{n!} \sum_{k = 0}^n \binom{n}{k} a^k b^{n-k} \\
        &= \dfrac{(a + b)^n}{n!}
    \end{align*}

    ainsi
    $$
    e^a e^b = \sum_{n \geq 0} \dfrac{(a + b)^n}{n!} 1^n = e^{a + b}
    $$
\end{demonstration}

\subsection{Développabilité en série entière}

\underline{On ne considèra maintenant que $z \in \R$.}

\begin{definition}[(Développabilité en série entière)]
    Soit $z_0 \in \R$ et $R \in \R_+ \cup \{+\infty\}$.

    Soit $f: I \to \R$ avec $]z_0 - R, z_0 + R[ \subseteq I$, on dira que $f$ sera \textbf{dévéloppable en série entière} sur $]z_0 - R, z_0 + R[$ si
    $$
    \forall z \in ]z_0 - R, z_0 + R[, \quad f(z) = \sum_{n \geq 0} a_n (z - z_0)^n
    $$
    cette somme s'appelle le \textbf{développement en série entière} de $f$ au voisinage de $z_0$.
\end{definition}

\begin{proposition}{}{}
    $f$ est développable en série entière sur $]z_0 - R, z_0 + R[$ si et seulement si
    $z \mapsto g(z + z_0)$ est développable en série entière sur $]-R, R[$ en $z = 0$.
\end{proposition}

\begin{theoreme}{Dérivation et intégration de fonctions d.s.e}{}
    Soit $f$ une fonction développable en série entière en 0, de rayon de convergence $R > 0$. 
    Pour tout $z \in ]-R, R[$, on a : 
    $$
    f(z) = \sum_{n=0}^{+\infty} a_n z^n
    $$

    Alors, $f$ est de classe $\mathcal{C}^\infty$ sur l'intervalle $]-R, R[$ et ses dérivées s'obtiennent par dérivation terme à terme. Pour tout $x \in ]-R, R[$ :
    $$
    f'(x) = \sum_{n=1}^{+\infty} n a_n x^{n-1}
    $$

    Plus généralement, pour tout entier $p \geq 1$
    $$
    f^{(p)}(x) = \sum_{n=p}^{+\infty} \frac{n!}{(n-p)!} a_n x^{n-p}
    $$

    Et on a aussi
    $$
    \int_0^x f(t) \,dt = \sum_{n=0}^{+\infty} \frac{a_n}{n + 1} x^{n + 1}
    $$
\end{theoreme}

\begin{corollaire}{}{}
    Si $f$ est d.s.e sur $]-R, R[$ avec $R > 0$.

    Alors
    $$
    f(z) = \sum_{n \geq 0} \dfrac{f^{(n)}(0)}{n!} z^n
    $$
\end{corollaire}

\begin{remarque}
    On reconnait un développement de Taylor de $f$ au voisinage de 0.
\end{remarque}

\begin{demonstration}
    On fait une récurrence sur $n$
\end{demonstration}

\begin{definition}
    Si $f$ est indéfiniment dérivable sur $]-R, R[$, alors le developpement,
    $$
    f(z) = \sum_{n \geq 0} \dfrac{f^{(n)}(0)}{n!} z^n
    $$
    s'appelle la série de Taylor de $f$ au voisinage de 0.
\end{definition}

\begin{proposition}{}{}
    Si $f$ est d.s.e sur $]-R, R[$, alors $f$ est indéfiniment dérivable sur $]-R, R[$ pour tout $n \in \N$.

    Elle admet un développement limité en 0 à l'ordre $n$, ça correspond à la somme partielle de la série de Taylor
    $$
    f(z) = \sum_{k = 0}^n \dfrac{f^{(k)}(0)}{k!} z^k + o(z^n)
    $$
\end{proposition}

\begin{corollaire}{}{}
    Soit $g$ une fonction continue sur $[a, b]$ où $-\infty < a < b < +\infty$.
    Soit $\funcdef{\varphi}{[0, 1]}{[a, b]}{t}{ta + (1 - t)b}$

    Si on pose $f = g \circ \varphi$, et si on note $P_n(\cdot, g) = B_n(\cdot, f \circ \varphi^{-1})$
    
    Alors $P_n(\cdot, g)$ converge uniformément vers $g$ sur $[a, b]$.
\end{corollaire}

\section{Les séries entières}

\begin{definition}
    Soit $(a_n)_{n \in \N}$ une suite de nombres réels ou complexe.
    On appelle \textbf{série entière} la série de fonctions de la forme
    $$
        \sum_{k = k_0}^{+\infty} a_k z^k
    $$
    où $z \in \mathbb{C}$.
\end{definition}

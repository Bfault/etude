\begin{definition}
    Soit $n, m \in \N$ avec $m \neq 0$.
    Un nombre $n$ est dit $m$-parasite si \text{$n \times m = d10^{L - 1} + \dfrac{n - d}{10}$} où $d$ est le chiffre des unités de $n$ et $L$ sa taille (nombre de chiffres).
\end{definition}

\begin{exemple}
    $$
    128205 \times 4 = 512820
    $$
\end{exemple}

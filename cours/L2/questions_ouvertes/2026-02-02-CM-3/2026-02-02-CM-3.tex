\subsection*{Travaux réalisés}

    \begin{itemize}[label=\textbullet]
        \item On donne une meilleure définition des nombres tournants et de leur longueur.
        \item Réponse à \textbf{[Q2]} : On a trouvé une formule pour trouver la longueur d'un nombre tournant à partir du couple $(m, d)$ (\textbf{Annexe B}).
        $$
        l = \text{ordre de } 10 \text{ dans } \Z/(\dfrac{10m - 1}{pgcd(d, 10m - 1)})\Z
        $$
        \item Réponse à \textbf{[Q1]} : La longueur dépend donc du multipliant et du pgcd du chiffre des unités (d'où certaines exceptions).
    \end{itemize}

\begin{definition}
    Soit $x, m \in \N^*$.
    Un nombre $x$ est dit \textbf{$m$-tournant} si \text{$x \times m = d10^{l - 1} + \dfrac{x - d}{10}$} où $d \in \llbracket 0, 9 \rrbracket$ est le chiffre des unités de $x$ et $l$ sa longueur.
\end{definition}

\begin{definition}
    La longueur d'un nombre $m$-tournant est le nombre de chiffres de ce nombre après rotation.
\end{definition}

\begin{remarque}
    Un nombre $m$-tournant peut être répété plusieurs fois et sera toujours un nombre $m$-tournant. Par exemple, $2$ est un nombre $10$-tournant de longueur $2$ et $202$ est aussi un nombre $10$-tournant de longueur $4$.
    Ainsi on ne s'intéressera qu'aux nombres tournants les plus petits possibles sans répétition.
\end{remarque}

\begin{proposition}{}{}
    Pour tout n-uplet $(x_1, \cdots, x_n)$ et tout $\varphi \in \wedge^{*n}(E)$, on a
    $$
    \forall u \in \mathcal{L}(E), \quad \varphi(u(x_1), \cdots, u(x_n)) = \det(u) \varphi(x_1, \cdots, x_n).
    $$
\end{proposition}

\begin{demonstration}
    On a nécessairement $\varphi = \lambda \Delta, k \in \K$ d'après le chapitre précédent.

    On a alors $\varphi(u(x_1), \cdots, u(x_n)) = \lambda \Delta(u(x_1), \cdots, u(x_n))$.

    Il suffit de démontrer la propriété pour $\varphi = \Delta$.

    Dans la base canonique, on a
    $$
    \forall j \in \llbracket 1, n \rrbracket, \quad x_j = \sum\limits_{i = 1}^n x_{i,j} e^i
    $$

    $X$ est la matrice des coordonnées de la famille $(x_1, \cdots, x_n)$ dans la base canonique.
    De même, $U = Mat_{Can}(u)$.
    On a alors
    $$
    y_j = U \begin{pmatrix}
        x_{1,j} \\
        \vdots  \\
        x_{n,j}
    \end{pmatrix}
    $$
    La matrice des coordonnées de la famille $(u(x_1), \cdots, u(x_n))$ dans la base canonique est donc $Y = UX$.
    On a donc
    $$
        \det(Y) = \det(U) \det(X)
    $$
\end{demonstration}

\begin{theoreme}{}{}
    \begin{enumerate}
        \item Le déterminant de l'application identité vaut 1.
        \item Si $u, v \in \L(E)$, alors $\det(u \circ v) = \det(u) \det(v)$.
        \item Un endomorphisme $u \in \L(E)$ est un isomorphisme si et seulement si $\det(u) \neq 0$. Dans ce cas, on a
        $$
        \det(u^{-1}) = \frac{1}{\det(u)}
        $$
    \end{enumerate}
\end{theoreme}

\begin{demonstration}
    Il suffit de démontrer les propriétés dans une base quelconque et d'utiliser les matrices associées.
\end{demonstration}

\begin{definition}
    Soit $A = (a_{i,j}) \in M_n(\K)$.
    \begin{enumerate}
        \item Le terme $(-1)^{i+j} \Delta_{i,j}$ est appelé \textbf{cofacteur} de $a_{i,j}$ on le note $A_{i,j}$.
        \item On appelle \textbf{comatrice} de $A$ la matrice des cofacteurs. On la note $com(A)$.
    \end{enumerate}
\end{definition}

\begin{theoreme}{}{}
    Soit $A \in M_n(\K)$. On a
    $$
    A \times com(A)^T = \det(A) Id_n = com(A)^T \times A
    $$
\end{theoreme}

\section{Réduction des endomorphismes}

Le but de la \textbf{réduction des endomorphismes} est de simplifier la représentation matricielle d'un endomorphisme en choisissant une base adaptée.\n
\vspace{1cm}

\begin{definition}
    Soit $u \in \L(E)$. On dit que $u$ est \textbf{diagonalisable} s'il existe une base de $E$ telle que la matrice de $u$ dans cette base est une matrice diagonale.
\end{definition}

\begin{definition}
    Soit $u \in \L(E)$ et $v \in E\setminus\{0_E\}$. On dit que $v$ est un \textbf{vecteur propre} de $u$ s'il existe $\lambda \in \K$ tel que
    $$
    u(v) = \lambda v
    $$
    Dans ce cas, on dit que $\lambda$ est une \textbf{valeur propre} de $u$ associée à $v$.
\end{definition}

\begin{remarque}
    Ainsi les vecteurs propres sont :
    \begin{itemize}
        \item des vecteurs de $\ker u$ associés à la valeur propre $0$ (s'ils existent)
        \item des vecteurs $v$ tels que $u(v)$ est colinéaire à $v$ (c'est-à-dire qui ne changent pas de direction par l'application de $u$)
    \end{itemize}
\end{remarque}

\begin{theoreme}{}{}
    $u \in \L(E)$ est diagonalisable si et seulement si il existe une base de $E$ constituée de vecteurs propres de $u$. 
\end{theoreme}

\begin{demonstration}
    $(\Rightarrow)$ Soit $\mathcal{B}$ une base de $E$ dans laquelle la matrice de $u$ est diagonale.
    
    ainsi la matrice de $u$ dans la base $\mathcal{B}$ est de la forme
    $$
    Mat_{\mathcal{B}}(u) = \begin{pmatrix}
        a_{11} & 0 & \cdots & 0 \\
        0 & a_{22} & \cdots & 0 \\
        \vdots & \vdots & \ddots & \vdots \\
        0 & 0 & \cdots & a_{nn}
    \end{pmatrix}
    $$
    où $a_{11}, \cdots, a_{nn} \in \K$.
    On a donc $u(e_i) = a_{ii} e_i$ pour tout $i \in \llbracket 1, n \rrbracket$.
    Ainsi, chaque vecteur de la base $\mathcal{B}$ est un vecteur propre de $u$.

    $(\Leftarrow)$ Réciproquement, soit $\mathcal{B} = (e_1, \cdots, e_n)$ une base de $E$ constituée de vecteurs propres de $u$.
    Pour tout $i \in \llbracket 1, n \rrbracket$, il existe $\lambda_i \in \K$ tel que $u(e_i) = \lambda_i e_i$.
    Ainsi, la matrice de $u$ dans la base $\mathcal{B}$ est de la forme
    $$
    Mat_{\mathcal{B}}(u) = \begin{pmatrix}
        \lambda_1 & 0 & \cdots & 0 \\
        0 & \lambda_2 & \cdots & 0 \\
        \vdots & \vdots & \ddots & \vdots \\
        0 & 0 & \cdots & \lambda_n
    \end{pmatrix}
    $$
    Ainsi, $u$ est diagonalisable.
\end{demonstration}
